       ```typescript
       export function escapeHtml(str: string): string {
         return str
           .replace(/&/g, "&amp;")
           .replace(/</g, "&lt;")
           .replace(/>/g, "&gt;")
           .replace(/"/g, "&quot;")
           .replace(/'/g, "&#039;")
       }
       ```

    2. In each integration controller:
       a. Add `import { escapeHtml } from "@lib/safe-html"`
       b. Add `import { pageLifecycle } from "@lib/page-lifecycle"`
       c. Wrap ALL interpolated server values in innerHTML with `escapeHtml()`:

          - `data.error` → `escapeHtml(data.error || "Unknown error")`
          - `err.message` → `escapeHtml(err instanceof Error ? err.message : "Unknown")`
          - `safeName` → `escapeHtml(safeName)`
          - `wf.id` → `escapeHtml(String(wf.id))`
          - `data.code || data.message` → `escapeHtml(data.code || data.message || "Unknown")`

       e. Add `{ signal: pageLifecycle.signal }` to all `fetch()` calls
       f. Guard DOM updates after fetch with: `if (pageLifecycle.signal.aborted) return`
